Status Rows = COUNTROWS ( 'TDOI Invoice Status' )

Status Label =
VAR n   = FORMAT ( [Status Rows], "#,0" )
VAR p   = FORMAT ( DIVIDE ( [Status Rows], CALCULATE ( [Status Rows], ALL ( 'Invoice Status Dim' ) ) ), "0.0%" )
VAR pad = UNICHAR ( 160 )
RETURN
    LEFT ( SELECTEDVALUE ( 'Invoice Status Dim'[status] ) & REPT ( pad, 18 ), 18 )
        & REPT ( pad, 8 - LEN ( n ) ) & n
        & REPT ( pad, 8 - LEN ( p ) ) & p

AGP Status Rows = COUNTROWS ( 'AGP Invoice Status' )

AGP Status Label =
VAR name  = SELECTEDVALUE ( 'AGP Accrual Status Dim'[status] )
VAR rows  = [AGP Status Rows]
VAR share = DIVIDE ( rows, CALCULATE ( [AGP Status Rows], ALL ( 'AGP Accrual Status Dim' ) ) )
VAR num   = FORMAT ( rows, "#,0" )
VAR pct   = IF ( share > 0 && share < 0.001, "<0.1%", FORMAT ( share, "0.0%" ) )
VAR pad   = UNICHAR ( 160 )
RETURN
    REPT ( pad, 15 - LEN ( name ) )
        & REPT ( pad, 9 - LEN ( num ) ) & num
        & REPT ( pad, 8 - LEN ( pct ) ) & pct
